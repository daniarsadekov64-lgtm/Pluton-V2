import os

# === НАСТРОЙКИ ВАШЕГО GITHUB ===
USERNAME = "daniarsadekov64-lgtm"       # Ваш ник со скриншота
REPO = "Pluton-V2"                     # Название вашего репозитория
BRANCH = "main"                         # Ветка
TOTAL_STEPS = 25                        # Количество шагов цепочки

# === ВАШ НАСТОЯЩИЙ СКРИПТ ROBLOX ===
CORE_SCRIPT = """
print("====================================")
print("Защита Pluton-V2 успешно пройдена!")
print("Ваш основной скрипт запущен.")
print("====================================")
"""

# ===================================================

output_dir = "chain_files"
os.makedirs(output_dir, exist_ok=True)

# 1. Создаем самый последний файл с вашим настоящим кодом
final_filename = "final_core.lua"
with open(os.path.join(output_dir, final_filename), "w", encoding="utf-8") as f:
    f.write(CORE_SCRIPT)

base_url = f"https://githubusercontent.com{USERNAME}/{REPO}/{BRANCH}/{output_dir}"

# 2. Генерируем цепочку файлов с конца к началу, чтобы ссылки выстраивались строго друг за другом
# Последний шаг (шаг 25) будет ссылаться на final_core.lua
next_target_url = f"{base_url}/{final_filename}"

for step in range(TOTAL_STEPS, 0, -1):
    # Текущий файл будет содержать прямую и полную ссылку на следующий файл
    step_code = f'loadstring(game:HttpGet("{next_target_url}"))()'
    
    filename = f"step_{step}.lua"
    with open(os.path.join(output_dir, filename), "w", encoding="utf-8") as f:
        f.write(step_code)
    
    # Теперь этот файл (например, step_25.lua) становится целью для предыдущего (step_24.lua)
    next_target_url = f"{base_url}/{filename}"

print("\n[УСПЕШНО] Все файлы цепочки сгенерированы!")
print("-" * 60)
print("ЧТО ДЕЛАТЬ СЕЙЧАС:")
print(f"1. Удалите старую папку 'chain_files' с компьютера и с GitHub.")
print(f"2. Загрузите НОВУЮ созданную папку 'chain_files' в ваш репозиторий '{REPO}'.")
print("3. Скопируйте и запустите в Roblox команду ниже (это полная ссылка на step_1.lua):\n")
print(f'loadstring(game:HttpGet("{base_url}/step_1.lua"))()')
print("-" * 60)
